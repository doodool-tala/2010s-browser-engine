```rust
//! The cookie jar: RFC 6265 storage, matching, and expiry, v1.
//! Normative reference: ADR-0012.
//!
//! Deterministic by construction: time is a parameter (`now`, epoch
//! seconds) supplied by the caller — never read from the OS. The jar
//! is pure state plus pure functions. Malformed Set-Cookie input is
//! RecoverableInput (ADR-0002): rejected silently, never an error.
//!
//! v1 subset (divergence register): a built-in public-suffix subset
//! (DV-009) instead of the full publicsuffix.org list; HttpOnly is
//! stored but not yet enforced (no non-HTTP accessor exists to
//! enforce against).

const DEFAULT_MAX_TOTAL: usize = 3000;
/// Default per-domain cap (RFC 6265's recommended minimum).

/// The parsed form of one Set-Cookie header value.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ParsedCookie {
    /// The cookie name.
    pub name: String,
    /// The cookie value.
    pub value: String,
    /// The Domain attribute, normalized (no leading dot, lowercase);
    /// empty attributes were ignored.
    pub domain: Option<String>,
    /// The Path attribute.
    pub path: Option<String>,
    /// The Max-Age attribute, in seconds.
    pub max_age: Option<i64>,
    /// The Expires attribute, in epoch seconds.
    pub expires: Option<u64>,
    /// The Secure flag.
    pub secure: bool,
    /// The HttpOnly flag.
    pub http_only: bool,
}

/// Parse one Set-Cookie header value (RFC 6265 §5.2). Returns `None`
/// when the string is not a valid cookie (RecoverableInput: the
/// caller silently drops it).
pub fn parse_set_cookie(value: &str) -> Option<ParsedCookie> {
    let mut parts = value.split(';');
    let pair = parts.next()?;
    let eq = pair.find('=')?;
    let name = pair[..eq].trim().to_string();
    if name.is_empty() {
        return None;
    }
    let cookie_value = pair[eq + 1..].trim().to_string();
    let mut parsed = ParsedCookie {
        name,
        value: cookie_value,
        domain: None,
        path: None,
        max_age: None,
        expires: None,
        secure: false,
        http_only: false,
    };
    for part in parts {
        let part = part.trim();
        if part.is_empty() {
            continue;
        }
        let (attr_name, attr_value) = match part.split_once('=') {
            Some((name, value)) => (name.trim().to_ascii_lowercase(), value.trim()),
            None => (part.to_ascii_lowercase(), ""),
        };
        match attr_name.as_str() {
            "domain" => {
                let domain = attr_value.trim_start_matches('.').to_ascii_lowercase();
                parsed.domain = (!domain.is_empty()).then_some(domain);
            }
            "path" => parsed.path = Some(attr_value.to_string()),
            "secure" => parsed.secure = true,
            "httponly" => parsed.http_only = true,
            "max-age" => {
                parsed.max_age = attr_value.parse::<i64>().ok();
            }
            "expires" => {
                parsed.expires = parse_http_date(attr_value);
            }
            _ => {}
        }
    }
    Some(parsed)
}

/// Parse an HTTP date (RFC 6265 §5.1.1 subset: IMF-fixdate and common
/// variants). Returns epoch seconds, or `None` when unparseable (the
/// attribute is then ignored, yielding a session cookie).
pub fn parse_http_date(value: &str) -> Option<u64> {
    let mut day: Option<u32> = None;
    let mut month: Option<u32> = None;
    let mut year: Option<i64> = None;
    let mut time: Option<(u32, u32, u32)> = None;
    for token in value.split(|c: char| !c.is_ascii_alphanumeric() && c != ':') {
        if token.is_empty() {
            continue;
        }
        if time.is_none() && token.contains(':') {
            let mut parts = token.split(':');
            let hour = parts.next()?.parse::<u32>().ok()?;
            let minute = parts.next()?.parse::<u32>().ok()?;
            let second = parts.next()?.parse::<u32>().ok()?;
            if hour > 23 || minute > 59 || second > 59 {
                return None;
            }
            time = Some((hour, minute, second));
        } else if token.chars().all(|c| c.is_ascii_digit()) {
            let number: i64 = token.parse().ok()?;
            if day.is_none() && (1..=31).contains(&number) && token.len() <= 2 {
                day = Some(number as u32);
            } else {
                year = Some(normalize_year(number));
            }
        } else if token.len() >= 3 {
            let month_token = token[..3].to_ascii_lowercase();
            if let Some(index) = MONTHS.iter().position(|m| *m == month_token) {
                month = Some(index as u32 + 1);
            }
        }
    }
    let day = day?;
    let month = month?;
    let year = year?;
    if !(1601..=9999).contains(&year) {
        return None;
    }
    let (hour, minute, second) = time?;
    let seconds =
        days_from_civil(year, month, day) * 86400 + (hour * 3600 + minute * 60 + second) as i64;
    if seconds < 0 {
        return None;
    }
    Some(seconds as u64)
}

fn normalize_year(number: i64) -> i64 {
    if (70..=99).contains(&number) {
        number + 1900
    } else if (0..=69).contains(&number) {
        number + 2000
    } else {
        number
    }
}

/// Days since the Unix epoch (Howard Hinnant's civil algorithm).
fn days_from_civil(year: i64, month: u32, day: u32) -> i64 {
    let y = if month <= 2 { year - 1 } else { year };
    let era = if y >= 0 { y } else { y - 399 } / 400;
    let yoe = y - era * 400;
    let m = month as i64;
    let doy = (153 * (if m > 2 { m - 3 } else { m + 9 }) + 2) / 5 + day as i64 - 1;
    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
    era * 146097 + doe - 719468
}

/// One stored cookie.
#[derive(Debug, Clone, PartialEq, Eq)]
struct Cookie {
    name: String,
    value: String,
    /// Host-only cookies match by equality; domain cookies carry the
    /// Domain attribute as `Host::Domain`.
    match_host: Host,
    domain_cookie: bool,
    path: String,
    secure: bool,
    http_only: bool,
    /// `None` means a session cookie.
    expires_at: Option<u64>,
    creation_seq: u64,
}

/// A cookie jar: store, match, and expire cookies per RFC 6265 v1.
pub struct CookieJar {
    cookies: Vec<Cookie>,
    next_seq: u64,
    max_total: usize,
    max_per_domain: usize,
}

impl CookieJar {
    /// A jar with the RFC's recommended minimum caps.
    #[must_use]
    pub fn new() -> Self {
        Self::with_caps(DEFAULT_MAX_TOTAL, DEFAULT_MAX_PER_DOMAIN)
    }

    /// A jar with explicit caps (tests and tight environments).
    #[must_use]
    pub fn with_caps(max_total: usize, max_per_domain: usize) -> Self {
        Self {
            cookies: Vec::new(),
            next_seq: 0,
            max_total,
            max_per_domain,
        }
    }

    /// Number of stored cookies.
    #[must_use]
    pub fn len(&self) -> usize {
        self.cookies.len()
    }

    /// True when nothing is stored.
    #[must_use]
    pub fn is_empty(&self) -> bool {
        self.cookies.is_empty()
    }

    /// Store one Set-Cookie value received for `url` at time `now`.
    /// Invalid or rejected cookies are silently dropped
    /// (RecoverableInput).
    pub fn store(&mut self, url: &Url, set_cookie_value: &str, now: u64) {
        let Some(parsed) = parse_set_cookie(set_cookie_value) else {
            return;
        };
        let (match_host, domain_cookie) = match &parsed.domain {
            None => (url.host().clone(), false),
            Some(domain) => match url.host() {
                Host::Domain(host) => {
                    if is_public_suffix(domain) || !domain_match(host, domain) {
                        return;
                    }
                    (Host::Domain(domain.clone()), true)
                }
                _ => return, // IP hosts cannot hold domain cookies
            },
        };
        let path = match &parsed.path {
            Some(path) if path.starts_with('/') => path.clone(),
            _ => default_path(url),
        };
        let expires_at = if let Some(max_age) = parsed.max_age {
            Some(now.saturating_add(max_age.max(0) as u64))
        } else {
            parsed.expires
        };
        if let Some(existing) = self.cookies.iter_mut().find(|cookie| {
            cookie.name == parsed.name && cookie.match_host == match_host && cookie.path == path
        }) {
            // Replacement keeps the original creation sequence.
            existing.value = parsed.value;
            existing.secure = parsed.secure;
            existing.http_only = parsed.http_only;
            existing.expires_at = expires_at;
            return;
        }
        self.next_seq += 1;
        self.cookies.push(Cookie {
            name: parsed.name,
            value: parsed.value,
            match_host,
            domain_cookie,
            path,
            secure: parsed.secure,
            http_only: parsed.http_only,
            expires_at,
            creation_seq: self.next_seq,
        });
        self.enforce_caps(now);
    }

    /// The Cookie header value for `url` at time `now`, or `None` when
    /// no cookie matches. Ordering per RFC 6265 §5.4: longer paths
    /// first, then earlier creation.
    pub fn cookie_header(&self, url: &Url, now: u64) -> Option<String> {
        let request_path = current_path(url);
        let mut matched: Vec<&Cookie> = self
            .cookies
            .iter()
            .filter(|cookie| self.matches(cookie, url, &request_path, now))
            .collect();
        if matched.is_empty() {
            return None;
        }
        matched.sort_by(|a, b| {
            b.path.len()
                .cmp(&a.path.len())
                .then(a.creation_seq.cmp(&b.creation_seq))
        });
        let header = matched
            .iter()
            .map(|cookie| format!("{}={}", cookie.name, cookie.value))
            .collect::<Vec<String>>()
            .join("; ");
        Some(header)
    }

    fn matches(&self, cookie: &Cookie, url: &Url, request_path: &str, now: u64) -> bool {
        if cookie.secure && !url.scheme().is_tls() {
            return false;
        }
        if let Some(expires_at) = cookie.expires_at {
            if now >= expires_at {
                return false;
            }
        }
        let host_matches = if cookie.domain_cookie {
            match (url.host(), &cookie.match_host) {
                (Host::Domain(host), Host::Domain(domain)) => domain_match(host, domain),
                _ => false,
            }
        } else {
            url.host() == &cookie.match_host
        };
        host_matches && path_match(request_path, &cookie.path)
    }

    fn enforce_caps(&mut self, now: u64) {
        self.cookies
            .retain(|cookie| cookie.expires_at.map_or(true, |at| now < at));
        while self.cookies.len() > self.max_total {
            self.evict_oldest();
        }
        loop {
            let over = self.cookies.iter().find_map(|cookie| {
                let count = self
                    .cookies
                    .iter()
                    .filter(|other| other.match_host == cookie.match_host)
                    .count();
                if count > self.max_per_domain {
                    Some(cookie.match_host.clone())
                } else {
                    None
                }
            });
            match over {
                Some(host) => self.evict_oldest_with_host(&host),
                None => break,
            }
        }
    }

    fn evict_oldest(&mut self) {
        if let Some(index) = self
            .cookies
            .iter()
            .enumerate()
            .min_by_key(|(_, cookie)| cookie.creation_seq)
            .map(|(index, _)| index)
        {
            self.cookies.remove(index);
        }
    }

    fn evict_oldest_with_host(&mut self, host: &Host) {
        if let Some(index) = self
            .cookies
            .iter()
            .enumerate()
            .filter(|(_, cookie)| &cookie.match_host == host)
            .min_by_key(|(_, cookie)| cookie.creation_seq)
            .map(|(index, _)| index)
        {
            self.cookies.remove(index);
        }
    }
}

impl Default for CookieJar {
    fn default() -> Self {
        Self::new()
    }
}

/// RFC 6265 §5.1.3 domain matching (string form; callers guarantee
/// the host is a domain, not an IP).
fn domain_match(host: &str, domain: &str) -> bool {
    if host == domain {
        return true;
    }
    host.ends_with(domain)
        && host.len() > domain.len()
        && host.as_bytes()[host.len() - domain.len() - 1] == b'.'
}

/// RFC 6265 §5.1.4 path matching.
fn path_match(request_path: &str, cookie_path: &str) -> bool {
    if request_path == cookie_path {
        return true;
    }
    request_path.starts_with(cookie_path)
        && (cookie_path.ends_with('/')
            || request_path.as_bytes().get(cookie_path.len()) == Some(&b'/'))
}

/// The request path of a URL, always starting with '/'.
fn current_path(url: &Url) -> String {
    let mut path = String::from("/");
    path.push_str(&url.path_segments().join("/"));
    path
}

/// The default cookie path per RFC 6265 §5.1.4: the request path's
/// directory.
fn default_path(url: &Url) -> String {
    let segments = url.path_segments();
    if segments.len() <= 1 {
        String::from("/")
    } else {
        let mut path = String::from("/");
        path.push_str(&segments[..segments.len() - 1].join("/"));
        path
    }
}

/// True when `domain` is in the built-in public-suffix subset.
fn is_public_suffix(domain: &str) -> bool {
    PUBLIC_SUFFIXES.contains(&domain)
}
```

```rust
    fn fetch(&mut self, url: &Url) -> Result<Resource, ModuleError> {
        let mut current = url.clone();
        let mut hops: u32 = 0;
        loop {
            let cookie_header = self
                .cookie_jar
                .as_ref()
                .and_then(|jar| jar.cookie_header(&current, self.epoch_secs));
            let (outcome, headers) =
                fetch_once(&current, &self.user_agent, cookie_header.as_deref())?;
            if let Some(jar) = &mut self.cookie_jar {
                for value in header_values(&headers, "set-cookie") {
                    jar.store(&current, value, self.epoch_secs);
                }
            }
            match outcome {
                Outcome::Body(resource) => return Ok(resource),
                Outcome::Redirect(location) => {
                    if hops >= self.max_redirects {
                        return Err(ModuleError::new("http", "too many redirects"));
                    }
                    hops += 1;
                    current = resolve_redirect(&current, &location)?;
                }
            }
        }
    }
```

```rust
fn fetch_once(
    url: &Url,
    user_agent: &str,
    cookie_header: Option<&str>,
) -> Result<(Outcome, Vec<(String, String)>), ModuleError> {
    if url.scheme().is_tls() {
        return Err(ModuleError::new(
            "http",
            "TLS not yet supported (DV-005); rustls arrives in a later package",
        ));
    }
    let port = url.port().unwrap_or_else(|| url.scheme().default_port());
    let connect_to = format!("{}:{port}", url.host());
    let host_header = match url.port() {
        Some(port) => format!("{}:{port}", url.host()),
        None => url.host().to_string(),
    };
    let target = request_target(url);
    let request = match cookie_header {
        Some(cookies) => format!(
            "GET {target} HTTP/1.1\r\nHost: {host_header}\r\nUser-Agent: {user_agent}\r\nAccept: */*\r\nAccept-Encoding: identity\r\nCookie: {cookies}\r\nConnection: close\r\n\r\n"
        ),
        None => format!(
            "GET {target} HTTP/1.1\r\nHost: {host_header}\r\nUser-Agent: {user_agent}\r\nAccept: */*\r\nAccept-Encoding: identity\r\nConnection: close\r\n\r\n"
        ),
    };
    let mut stream = TcpStream::connect(connect_to.as_str())
        .map_err(|e| ModuleError::new("http", format!("connect failed: {e}")))?;
    stream
        .write_all(request.as_bytes())
        .map_err(|e| ModuleError::new("http", format!("request write failed: {e}")))?;
    stream
        .flush()
        .map_err(|e| ModuleError::new("http", format!("request flush failed: {e}")))?;
    let mut reader = BufReader::new(stream);
    let status = parse_status(&read_crlf_line(&mut reader)?)?;
    let headers = parse_headers(&mut reader)?;
    let body = read_body(&mut reader, &headers)?;
    if (200..300).contains(&status) {
        let content_type = header_value(&headers, "content-type").map(str::to_string);
        Ok((Outcome::Body(Resource::new(body, content_type)), headers))
    } else if (300..400).contains(&status) {
        match header_value(&headers, "location") {
            Some(location) => Ok((Outcome::Redirect(location.to_string()), headers)),
            None => Err(ModuleError::new(
                "http",
                "redirect without a location header",
            )),
        }
    } else {
        Err(ModuleError::new("http", format!("status {status}")))
    }
}
```

```markdown
# ADR-0012: Cookie jar v1
Status: accepted

## Decision
- The jar implements RFC 6265's v1 subset: §5.2 Set-Cookie parsing,
  §5.1.3 domain matching, §5.1.4 path matching, §5.3 storage rules
  (host-only vs Domain cookies, leading-dot normalization, default
  paths), §5.4 retrieval ordering (longer paths, then earlier
  creation), and Max-Age-over-Expires precedence.
- Time is a parameter: every store/retrieve takes `now` in epoch
  seconds. Wall-clock never enters engine code (ADR-0003); the
  browser process supplies real time at its boundary, tests supply
  fixed values, and the default is 0.
- The jar lives INSIDE HttpTransport (the v1 Transport seam cannot
  carry outgoing headers, so a loader-level jar would be dead code).
  Consequences: zero contract amendments for the seam, cookies
  absorb naturally at every redirect hop, and the architecture
  matches Chrome's — the network stack owns the cookie store.
  C-025 gains two additive builder methods (v2).
- Caps: 3000 total, 50 per domain; on every store, expired cookies
  are swept first, then the oldest by creation sequence. Determinism
  holds throughout: eviction order is a pure function of inputs.
- Malformed Set-Cookie is RecoverableInput (ADR-0002): silently
  dropped, never an error, never a crash.
- Public-suffix checking uses a built-in subset (DV-009). HttpOnly is
  stored but not enforced — no non-HTTP accessor exists yet.

## Consequences
- The jar's behavior is locked byte-exact by the golden test; the
  integration test proves cookies flow over real sockets.
- document.cookie (later, with the DOM) will read this jar through
  the eventual renderer↔browser IPC.
- Session cookies are memory-only by design; persistence is a later
  package.
```
